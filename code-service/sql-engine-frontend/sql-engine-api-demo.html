基础配置

数据源通过请求头 X-DB-Name 指定，不传走默认主库。
const SQL_ENGINE_BASE = '/api/sql-engine';

function buildHeaders(dbName) {
  const headers = { 'Content-Type': 'application/json' };
  if (dbName) headers['X-DB-Name'] = dbName;
  return headers;
}

async function callSqlEngine(endpoint, body, dbName) {
  const resp = await fetch(`${SQL_ENGINE_BASE}${endpoint}`, {
    method: 'POST',
    headers: buildHeaders(dbName),
    body: JSON.stringify(body)
  });
  if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
  return resp.json();
}

1. 单条查询 POST /api/sql-engine/query

带条件 + 分页 + 排序

const result = await callSqlEngine('/query', {
  tableName: 'sys_user',
  columns: ['id', 'name', 'age', 'status', 'create_time'],
  conditions: [
    { Symbol: 'and' },
    { Id: 'name', Symbol: 'like', Val: '张', TableAlias: 'T' },
    { Symbol: 'and' },
    { Id: 'age', Symbol: '>', Val: '18', TableAlias: 'T' },
    { Symbol: 'or' },
    { Id: 'status', Symbol: '=', Val: 'active', TableAlias: 'T' }

  orderBy: [{ column: 'create_time', direction: 'desc' }],
  page: 1,
  pageSize: 20
}, 'default');

// result = [{ id: 1, name: '张三', age: 25, ... }, ...]

3. 单条新增 POST /api/sql-engine/insert

const result = await callSqlEngine('/insert', {
  tableName: 'sys_user',
  data: {
    name: '张三',
    age: 25,
    status: 'active',
    email: 'zhangsan@example.com'
  }
});
// result = { success: true, affectedRows: 1 }

可替换属性：序列主键 + 数据库时间 + 跳过列

INSERT 时主键由数据库序列生成、审计字段用数据库时间、自增字段跳过由 DB 处理。
// Oracle 场景：主键用 SEQ_USERS.NEXTVAL
const result = await callSqlEngine('/insert', {
  tableName: 'sys_user',
  data: {
    name: '张三',
    age: 25
  },
  // 序列主键：列名 → 序列名，后端按方言生成 NEXTVAL
  sequenceColumns: {
    id: 'SEQ_USERS'
  },
  // 数据库时间自动填充（编译为 CURRENT_TIMESTAMP / SYSTIMESTAMP / SYSDATE / GETDATE()）
  autoTimestampColumns: ['create_time', 'update_time'],
  // 跳过列：完全不出现在 INSERT 中，由 DB 默认值/触发器处理
  skipColumns: ['tenant_id']
});
// 编译结果（Oracle）:
// INSERT INTO sys_user (id, name, age, create_time, update_time)
// VALUES (SEQ_USERS.NEXTVAL, ?, ?, SYSTIMESTAMP, SYSTIMESTAMP)

// PostgreSQL 场景：
// sequenceColumns: { id: 'seq_users' }
// 编译结果: nextval('seq_users')

可替换属性：自定义原始表达式

const result = await callSqlEngine('/insert', {
  tableName: 'sys_user',
  data: { name: '张三' },
  rawExpressions: {
    version: '1',                    // 字面量
    guid: "rawtohex(sys_guid())"     // Oracle GUID 生成
  }
});
// 编译结果:
// INSERT INTO sys_user (name, version, guid)
// VALUES (?, 1, rawtohex(sys_guid()))

9. 批量查询 POST /api/sql-engine/batch-query

多条 SELECT 在同一连接执行，返回多结果集。适用于 Dashboard 首页：一次请求拿多个统计数据。
const result = await callSqlEngine('/batch-query', {
  queries: [
    // 查询 1：今日新增用户
    {
      tableName: 'sys_user',
      columns: ['id', 'name', 'create_time'],
      conditions: [
        { Symbol: 'and' },
        { Id: 'create_time', Symbol: '>=', Val: '2026-09-09 00:00:00', TableAlias: 'T' }
      ],
      orderBy: [{ column: 'create_time', direction: 'desc' }],
      pageSize: 10
    },
    // 查询 2：待处理订单
    {
      tableName: 'biz_order',
      columns: ['order_no', 'status', 'amount'],
      conditions: [
        { Symbol: 'and' },
        { Id: 'status', Symbol: '=', Val: 'pending', TableAlias: 'T' }
      ],
      pageSize: 10
    },
    // 查询 3：库存预警
    {
      tableName: 'wms_inventory',
      columns: ['sku', 'warehouse', 'qty'],
      conditions: [
        { Symbol: 'and' },
        { Id: 'qty', Symbol: '<', Val: '10', TableAlias: 'T' }
      ],
      pageSize: 20
    }
  ]
});

// result[0] → 今日新增用户 JSONArray
// result[1] → 待处理订单 JSONArray
// result[2] → 库存预警 JSONArray

条件 JSON 格式速查表

操作符含义JSON 示例
=等于{"Id":"name","Symbol":"=","Val":"张三"}
<>不等于{"Id":"status","Symbol":"<>","Val":"0"}
<小于{"Id":"age","Symbol":"<","Val":"18"}
<=小于等于{"Id":"age","Symbol":"<=","Val":"18"}
>大于{"Id":"age","Symbol":">","Val":"18"}
>=大于等于{"Id":"age","Symbol":">=","Val":"18"}
like模糊匹配（自动加%）{"Id":"name","Symbol":"like","Val":"张"}
is null为空{"Id":"email","Symbol":"is null"}
is not null非空{"Id":"email","Symbol":"is not null"}
in包含（逗号分隔）{"Id":"dept","Symbol":"in","Val":"1,2,3"}
between区间（逗号分隔）{"Id":"age","Symbol":"between","Val":"18,30"}
raw自定义SQL模板（列名+SQL）{"Id":"status","Symbol":"raw","Val":"IN ('active','pending')"}
rpcSQL片段注入（完整表达式）{"Id":"expr","Symbol":"rpc","Val":"EXISTS(SELECT 1 FROM orders WHERE orders.user_id = T.id)"}

Rps / Rpc：自定义 SQL 条件

当 Op 枚举无法表达的复杂条件，用 raw 或 rpc 直接注入 SQL 片段。
// Rps: 列名 + 自定义 SQL 模板 → WHERE T.STATUS IN ('active','pending')
conditions: [
  {Symbol: "and"},
  {Id: "name", Symbol: "like", Val: "张", TableAlias: "T"},
  {Symbol: "and"},
  {Id: "status", Symbol: "raw", Val: "IN ('active','pending')", TableAlias: "T"}
]

// Rpc: 完整 SQL 片段 → WHERE T.ID = ? AND EXISTS(...)
conditions: [
  {Symbol: "and"},
  {Id: "id", Symbol: "=", Val: 100, TableAlias: "T"},
  {Symbol: "and"},
  {Id: "expr", Symbol: "rpc", Val: "EXISTS(SELECT 1 FROM orders WHERE orders.user_id = T.id)"}
]

条件组合规则

交替排列：[连接符, 条件项, 连接符, 条件项, ...]
// (name LIKE '张%' AND age > 18) OR status = 'active'
[
  { Symbol: 'and' },

完整业务场景：用户管理 CRUD

从搜索表单构建条件 → 查询 + 计数 → 新增 → 修改 → 删除 完整流程。
// 从搜索表单构建条件 JSON
function buildConditions(searchForm) {
  const conditions = [{ Symbol: 'and' }];
  if (searchForm.name) {
    conditions.push({
      Id: 'name', Symbol: 'like', Val: searchForm.name, TableAlias: 'T'
    });
    conditions.push({ Symbol: 'and' });
  }
  if (searchForm.status) {
    conditions.push({
      Id: 'status', Symbol: '=', Val: searchForm.status, TableAlias: 'T'
    });
  }
  return conditions.length > 1 ? conditions : [];
}

// 查询 + 分页（并行请求）
async function loadUsers(page = 1, pageSize = 20) {
  const conditions = buildConditions(searchForm);
  const [data, countResult] = await Promise.all([
    callSqlEngine('/query', {
      tableName: 'sys_user',
      columns: ['id', 'name', 'status', 'create_time'],
      conditions,
      orderBy: [{ column: 'create_time', direction: 'desc' }],
      page, pageSize
    }),
    callSqlEngine('/count', {
      tableName: 'sys_user',
      conditions
    })
  ]);
  // data → 当前页数据
  // countResult.total → 总记录数 → 计算总页数
  return { data, total: countResult.total };
}

// 批量修改（状态批量变更）
async function batchUpdateStatus(ids, status) {
  return callSqlEngine('/batch-update', {
    tableName: 'sys_user',
    batch: ids.map(id => ({
      data: { status },
      conditions: [{ Id: 'id', Symbol: '=', Val: String(id), TableAlias: 'T' }]
    }))
  });
}

Vue 3 Composition API

// <script setup>
import { ref, onMounted } from 'vue'

const users = ref([])
const total = ref(0)
const loading = ref(false)

async function loadUsers(page = 1) {
  loading.value = true
  try {
    const [data, countResult] = await Promise.all([
      callSqlEngine('/query', {
        tableName: 'sys_user',
        columns: ['id', 'name', 'status'],
        conditions: buildConditions(),
        page,
        pageSize: 20
      }),
      callSqlEngine('/count', {
        tableName: 'sys_user',
        conditions: buildConditions()
      })
    ])
    users.value = data
    total.value = countResult.total
  } finally {
    loading.value = false
  }
}

onMounted(() => loadUsers())
// </script>

React Hooks

import { useState, useEffect, useCallback } from 'react'

function useSqlQuery(tableName, conditions) {
  const [data, setData] = useState([])
  const [total, setTotal] = useState(0)
  const [loading, setLoading] = useState(false)

  const load = useCallback(async (page = 1) => {
    setLoading(true)
    try {
      const [rows, countResult] = await Promise.all([
        callSqlEngine('/query', {
          tableName,
          conditions,
          page,
          pageSize: 20,
          orderBy: [{ column: 'id', direction: 'desc' }]
        }),
        callSqlEngine('/count', { tableName, conditions })
      ])
      setData(rows)
      setTotal(countResult.total)
    } finally {
      setLoading(false)
    }
  }, [tableName, conditions])

  useEffect(() => { load(1) }, [])
  return { data, total, loading, load }
}
